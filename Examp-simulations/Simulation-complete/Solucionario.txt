5. ConfigMap app-config + Pod que lo use
bash
kubectl create configmap app-config \
  --from-literal=MODE=prod \
  --from-literal=VERSION=1.0 \
  -n production
app-pod.yaml:

yaml
apiVersion: v1
kind: Pod
metadata:
  name: app-pod
  namespace: production
spec:
  containers:
  - name: app
    image: busybox
    command: ["sh", "-c", "env; sleep 3600"]
    envFrom:
    - configMapRef:
        name: app-config
bash
kubectl apply -f app-pod.yaml

7. PV data-pv, PVC data-pvc y Pod
pv.yaml:

yaml
apiVersion: v1
kind: PersistentVolume
metadata:
  name: data-pv
spec:
  capacity:
    storage: 2Gi
  accessModes:
  - ReadWriteOnce
  hostPath:
    path: /data-pv
pvc.yaml:

yaml
apiVersion: v1
kind: PersistentVolumeClaim
metadata:
  name: data-pvc
  namespace: production
spec:
  accessModes:
  - ReadWriteOnce
  resources:
    requests:
      storage: 2Gi
pv-pod.yaml:

yaml
apiVersion: v1
kind: Pod
metadata:
  name: pv-pod
  namespace: production
spec:
  containers:
  - name: app
    image: busybox
    command: ["sh", "-c", "echo 'hola CKA' > /data/hola.txt; sleep 3600"]
    volumeMounts:
    - name: data
      mountPath: /data
  volumes:
  - name: data
    persistentVolumeClaim:
      claimName: data-pvc
bash
kubectl apply -f pv.yaml
kubectl apply -f pvc.yaml
kubectl apply -f pv-pod.yaml

10. Ingress Controller + web-ingress
Ejemplo usando NGINX Ingress Controller:

bash
kubectl apply -f https://raw.githubusercontent.com/kubernetes/ingress-nginx/main/deploy/static/provider/cloud/deploy.yaml
web-ingress.yaml (en el namespace donde estén los services app1-svc y app2-svc, supongamos production):

yaml
apiVersion: networking.k8s.io/v1
kind: Ingress
metadata:
  name: web-ingress
  namespace: production
spec:
  rules:
  - http:
      paths:
      - path: /app1
        pathType: Prefix
        backend:
          service:
            name: app1-svc
            port:
              number: 80
      - path: /app2
        pathType: Prefix
        backend:
          service:
            name: app2-svc
            port:
              number: 80
bash
kubectl apply -f web-ingress.yaml


11. ServiceAccount viewer-sa con permisos para listar Pods
bash
kubectl create sa viewer-sa -n production
role-viewer.yaml:

yaml
apiVersion: rbac.authorization.k8s.io/v1
kind: Role
metadata:
  name: pod-viewer
  namespace: production
rules:
- apiGroups: [""]
  resources: ["pods"]
  verbs: ["get", "list"]
rb-viewer.yaml:

yaml
apiVersion: rbac.authorization.k8s.io/v1
kind: RoleBinding
metadata:
  name: viewer-binding
  namespace: production
subjects:
- kind: ServiceAccount
  name: viewer-sa
  namespace: production
roleRef:
  kind: Role
  name: pod-viewer
  apiGroup: rbac.authorization.k8s.io
bash
kubectl apply -f role-viewer.yaml
kubectl apply -f rb-viewer.yaml
kubectl auth can-i list pods --as=system:serviceaccount:production:viewer-sa -n production


12. Pod health-pod con liveness y readiness
yaml
apiVersion: v1
kind: Pod
metadata:
  name: health-pod
  namespace: production
spec:
  containers:
  - name: app
    image: nginx
    ports:
    - containerPort: 80
    livenessProbe:
      httpGet:
        path: /healthz
        port: 80
      initialDelaySeconds: 10
      periodSeconds: 10
    readinessProbe:
      httpGet:
        path: /ready
        port: 80
      initialDelaySeconds: 5
      periodSeconds: 5
bash
kubectl apply -f health-pod.yaml
(en la práctica, deberías servir esas rutas; en el examen muchas veces solo piden definir la probe)

16. Deployment con topologySpreadConstraints
yaml
apiVersion: apps/v1
kind: Deployment
metadata:
  name: spread-api
  namespace: production
spec:
  replicas: 6
  selector:
    matchLabels:
      app: spread-api
  template:
    metadata:
      labels:
        app: spread-api
    spec:
      topologySpreadConstraints:
      - maxSkew: 1
        topologyKey: topology.kubernetes.io/zone
        whenUnsatisfiable: DoNotSchedule
        labelSelector:
          matchLabels:
            app: spread-api
      containers:
      - name: app
        image: nginx
bash
kubectl apply -f spread-api.yaml

18. Troubleshooting broken-app (CrashLoopBackOff)
Pasos generales:

bash
kubectl get pod broken-app -n production
kubectl describe pod broken-app -n production
kubectl logs broken-app -n production
Ejemplos de fixes típicos:

Imagen incorrecta:

bash
kubectl set image pod/broken-app app=nginx:1.25 -n production
Comando mal → kubectl edit pod broken-app -n production y corriges command / args.

Env faltante → agregas variables o Secret/ConfigMap.

Lo importante en el examen es mostrar que identificas la causa con logs + describe y luego aplicas el fix.
